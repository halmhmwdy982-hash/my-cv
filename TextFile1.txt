import json
class Person:
    system_name = "Car Maintenance Management System"
    def __init__(self, person_id, username, phone, email):
        self._person_id = person_id
        self._username = username
        self._phone = phone
        self._email = email
    @property
    def username(self):
        return self._username

    @username.setter
    def username(self, value):

        if not value.strip():
            raise ValueError("Username cannot be empty")
        self._username = value

    @property
    def phone(self):
        return self._phone
    @phone.setter
    def phone(self, value):
        Person.validate_phone(value)
        self._phone = value
        
    @property
    def email(self):
        return self._email
    @email.setter
    def email(self, value):
        if "@" not in value or "." not in value:
            raise ValueError("Invalid email")
        self._email = value

    def display_info(self):
        return (
            f"ID: {self._person_id} | "
            f"Username: {self._username} | "
            f"Phone: {self._phone} | "
            f"Email: {self._email}"
        )
    @staticmethod
    def validate_phone(phone):
        if not phone.isdigit():
            raise ValueError(    "Phone must contain numbers only")
        if len(phone) != 9:
            raise ValueError("Phone must contain 9 digits")
        if not phone.startswith(("77", "78", "73", "71")):
            raise ValueError("Phone must start with 77, 78, 73 or 71")
        return True
    @classmethod
    def get_system_name(cls):
        return cls.system_name

    def __str__(self):
        return f"{self._person_id} - {self._username}"
    def __eq__(self, other):
        if isinstance(other, Person):
            return self._person_id == other._person_id
        return False

class Vehicle:
    vehicle_types = { "1": "Toyota","2": "Hyundai","3": "Kia","4": "Nissan","5": "Other"
    }
    def __init__(self,vehicle_id,plate_number,model,year,owner_id):
        self.vehicle_id = vehicle_id
        self.plate_number = plate_number
        self.model = model
        self.year = year
        self.owner_id = owner_id

    def display_info(self):
        return (
            f"Vehicle ID: {self.vehicle_id} | "
            f"Plate: {self.plate_number} | "
            f"Model: {self.model} | "
            f"Year: {self.year} | "
            f"Owner ID: {self.owner_id}"        )

    @staticmethod
    def validate_year(year):
        if not str(year).isdigit():
            raise ValueError( "Year must contain numbers only")
        year = int(year)
        if year < 1980 or year > 2026:
            raise ValueError("Year must be between 1980 and 2026")
        return True
    def __str__(self):
        return (f"{self.vehicle_id} - {self.model}")
